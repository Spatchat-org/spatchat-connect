# Standalone reproducibility script for SpatChat graph outputs
# Install requirements first:
#   pip install pandas numpy networkx rasterio pyproj scikit-image
# Rename this file to .py if you want to run it directly.
#
# USER ARGUMENTS:
# - NODES_CSV: defaults to bundled nodes.csv or repro_config.json nodes_csv.
# - COST_TIF: defaults to bundled analysis_cost_tif/input_raster from repro_config.json.
# - OUTPUT_DIR: where reproduced outputs will be written.
# - GRAPH PARAMS: metric choices and weighting can be edited in GraphAnalysisParams.
# - SELECTED METRIC META: defaults to repro_config.json graph.selected_metric.

import json
import math
import os
from dataclasses import dataclass, field
from typing import Optional, Sequence

import networkx as nx
import numpy as np
import pandas as pd
import rasterio
from pyproj import CRS, Transformer
from rasterio.transform import rowcol
from rasterio.windows import Window
from skimage.graph import MCP_Geometric

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
CONFIG_PATH = os.path.join(SCRIPT_DIR, "repro_config.json")
OUTPUT_DIR = os.path.join(SCRIPT_DIR, "reproduced_outputs", "graph")


def _xy_to_rc(transform, x, y):
    r, c = rowcol(transform, x, y, op=math.floor)
    return int(r), int(c)


def _estimate_pixel_size(transform):
    sx = math.hypot(transform.a, transform.b)
    sy = math.hypot(transform.d, transform.e)
    return float((sx + sy) / 2.0) if sx and sy else float(sx or sy)


def _union_crop_window(points_rc, h, w, margin):
    r_vals = [p[0] for p in points_rc]
    c_vals = [p[1] for p in points_rc]
    return Window.from_slices((max(0, min(r_vals) - margin), min(h, max(r_vals) + margin)), (max(0, min(c_vals) - margin), min(w, max(c_vals) + margin)))


def compute_lcp_index(cost_tif, nodes_df, out_csv, id_col="node_id", lon_col="longitude", lat_col="latitude", nodes_crs="EPSG:4326"):
    with rasterio.open(cost_tif) as ds:
        H, W, tr, crs = ds.height, ds.width, ds.transform, ds.crs
        px = _estimate_pixel_size(tr)
        if nodes_crs and crs and CRS.from_user_input(nodes_crs) != CRS.from_user_input(crs):
            to_raster = Transformer.from_crs(nodes_crs, crs, always_xy=True)
        else:
            to_raster = None
        ids, rc_lookup = [], {}
        for _, row in nodes_df.iterrows():
            nid = str(row[id_col])
            x, y = float(row[lon_col]), float(row[lat_col])
            if to_raster is not None:
                x, y = to_raster.transform(x, y)
            r, c = _xy_to_rc(tr, x, y)
            if 0 <= r < H and 0 <= c < W:
                ids.append(nid)
                rc_lookup[nid] = (r, c)
        if len(ids) < 2:
            raise ValueError("Need at least two valid nodes")
        rows = []
        for i, src in enumerate(ids):
            src_rc = rc_lookup[src]
            tgts = [(ids[j], rc_lookup[ids[j]]) for j in range(i + 1, len(ids))]
            if not tgts:
                continue
            dmax = 0.0
            for _, rc in tgts:
                dr, dc = abs(rc[0] - src_rc[0]), abs(rc[1] - src_rc[1])
                dmax = max(dmax, (dr * dr + dc * dc) ** 0.5)
            win = _union_crop_window([src_rc] + [rc for _, rc in tgts], H, W, max(256, int(128 + 0.2 * dmax)))
            arr = np.array(ds.read(1, window=win, masked=True).filled(np.nan), dtype=np.float64)
            if np.isnan(arr).all():
                continue
            arr = np.where(np.isfinite(arr), arr, 1e12) * px
            r_off, c_off = int(win.row_off), int(win.col_off)
            start = (src_rc[0] - r_off, src_rc[1] - c_off)
            ends = [(rc[0] - r_off, rc[1] - c_off) for _, rc in tgts]
            mcp = MCP_Geometric(arr, fully_connected=True)
            costs, _ = mcp.find_costs(starts=[start], ends=ends)
            for (dst, _), end in zip(tgts, ends):
                cval = float(costs[end[0], end[1]])
                if np.isfinite(cval):
                    rows.append({"source": src, "target": dst, "cost": cval})
        pd.DataFrame(rows).to_csv(out_csv, index=False)
    return out_csv


def build_probability_decay_graph(edges: pd.DataFrame):
    if "cost" not in edges.columns:
        return edges.copy(), {"mode": "none", "edge_count_total": int(len(edges)), "edge_count_kept": int(len(edges))}
    decayed = edges.copy()
    costs = pd.to_numeric(decayed["cost"], errors="coerce")
    valid = costs[costs.notna() & (costs > 0)]
    if valid.empty:
        decayed["probability"] = 1.0
        return decayed, {"mode": "flat", "edge_count_total": int(len(decayed)), "edge_count_kept": int(len(decayed)), "probability_threshold": 0.0}
    scale = float(valid.median())
    decayed["probability"] = np.exp(-costs / max(scale, 1e-12))
    thr = min(0.2, max(0.05, float(decayed["probability"].median())))
    kept = decayed[decayed["probability"] >= thr].copy()
    if kept.empty:
        kept = decayed.nlargest(min(10, len(decayed)), "probability").copy()
        thr = float(kept["probability"].min())
    return kept, {"mode": "exponential_decay", "decay_scale_cost": scale, "probability_threshold": float(thr), "edge_count_total": int(len(decayed)), "edge_count_kept": int(len(kept))}


@dataclass(slots=True)
class GraphAnalysisParams:
    source_col: str = "source"
    target_col: str = "target"
    distance_col: Optional[str] = None
    strength_col: Optional[str] = None
    directed: bool = False
    metrics: Sequence[str] = field(default_factory=lambda: ("degree", "betweenness", "closeness", "pagerank", "clustering", "components"))


def run_graph_theory_analysis(edges: pd.DataFrame, params: GraphAnalysisParams):
    G = nx.DiGraph() if params.directed else nx.Graph()
    use_dist = params.distance_col and params.distance_col in edges.columns
    use_str = params.strength_col and params.strength_col in edges.columns
    for _, row in edges.iterrows():
        attrs = {}
        if use_dist:
            try:
                v = float(row[params.distance_col])
                if np.isfinite(v):
                    attrs["distance"] = v
            except Exception:
                pass
        if use_str:
            try:
                v = float(row[params.strength_col])
                if np.isfinite(v):
                    attrs["strength"] = v
            except Exception:
                pass
        G.add_edge(row[params.source_col], row[params.target_col], **attrs)
    node_df = pd.DataFrame(index=pd.Index(sorted(G.nodes()), name="node"))
    done = []
    for m in tuple(dict.fromkeys(params.metrics)):
        if m == "degree":
            node_df["degree"] = pd.Series(dict(G.degree(weight="strength" if use_str else None)))
            done.append(m)
        elif m == "betweenness":
            node_df["betweenness"] = pd.Series(nx.betweenness_centrality(G, weight="distance" if use_dist else None, normalized=True))
            done.append(m)
        elif m == "closeness":
            node_df["closeness"] = pd.Series(nx.closeness_centrality(G, distance="distance" if use_dist else None))
            done.append(m)
        elif m == "pagerank":
            node_df["pagerank"] = pd.Series(nx.pagerank(G, weight="strength" if use_str else None))
            done.append(m)
        elif m == "clustering":
            node_df["clustering"] = pd.Series(nx.clustering(G.to_undirected() if params.directed else G, weight="strength" if use_str else None))
            done.append(m)
        elif m == "components":
            comps = list(nx.weakly_connected_components(G) if params.directed else nx.connected_components(G))
            cid, csz = {}, {}
            for i, nodes in enumerate(comps):
                for n in nodes:
                    cid[n] = i
                    csz[n] = len(nodes)
            node_df["component_id"] = pd.Series(cid)
            node_df["component_size"] = pd.Series(csz)
            done.append(m)
    out = {"node_metrics": node_df.sort_index(), "metrics": tuple(done)}
    if "components" in done:
        out["component_count"] = int(node_df["component_id"].nunique())
    return out


with open(CONFIG_PATH, "r", encoding="utf-8") as f:
    CONFIG = json.load(f)
os.makedirs(OUTPUT_DIR, exist_ok=True)

NODES_CSV = os.path.join(SCRIPT_DIR, CONFIG.get("nodes_csv") or "nodes.csv")
COST_TIF = os.path.join(SCRIPT_DIR, CONFIG.get("analysis_cost_tif") or CONFIG.get("input_raster") or "input_raster.tif")
LON_COL = CONFIG.get("nodes_lon_col") or "longitude"
LAT_COL = CONFIG.get("nodes_lat_col") or "latitude"
nodes_df = pd.read_csv(NODES_CSV)

lcp_dir = os.path.join(OUTPUT_DIR, "graph_lcp")
os.makedirs(lcp_dir, exist_ok=True)
lcp_index_csv = compute_lcp_index(
    cost_tif=COST_TIF,
    nodes_df=nodes_df,
    out_csv=os.path.join(lcp_dir, "lcp_index.csv"),
    id_col="node_id",
    lon_col=LON_COL,
    lat_col=LAT_COL,
)
edges = pd.read_csv(lcp_index_csv)
filtered_edges, decay_summary = build_probability_decay_graph(edges)
result = run_graph_theory_analysis(filtered_edges, GraphAnalysisParams(distance_col="cost", strength_col="probability"))

metrics_path = os.path.join(OUTPUT_DIR, "graph_node_metrics.csv")
result["node_metrics"].reset_index().to_csv(metrics_path, index=False)
summary_path = os.path.join(OUTPUT_DIR, "graph_summary.json")
with open(summary_path, "w", encoding="utf-8") as f:
    json.dump(
        {
            "metrics": list(result.get("metrics") or ()),
            "component_count": result.get("component_count"),
            "probability_decay": decay_summary,
            "selected_metric": (CONFIG.get("graph") or {}).get("selected_metric"),
        },
        f,
        indent=2,
    )
print(f"Graph outputs reproduced in: {OUTPUT_DIR}")
print(f"Metrics written to: {metrics_path}")
